import { db } from "@/db/client";
import { generateWeeklyContent } from "@/lib/content-engine";

export async function POST(req) {
  const { contact_id } = await req.json();
  const database = await db();

  const contact = await database.get("SELECT * FROM contacts WHERE id = ?", contact_id);

  const result = await generateWeeklyContent(contact.brand_affinity);

  await database.run(
    "INSERT INTO generated_assets (contact_id, type, script, created_at) VALUES (?, ?, ?, datetime('now'))",
    contact_id, "script", result.script
  );

  return Response.json(result);
}

import { db } from "@/db/client";
import { generateVoiceover } from "@/lib/elevenlabs";
import { renderTrapVideo } from "@/video";

export async function POST(req) {
  const { contact_id } = await req.json();
  const database = await db();

  const asset = await database.get(
    "SELECT * FROM generated_assets WHERE contact_id = ? ORDER BY id DESC LIMIT 1",
    contact_id
  );

  const audioPath = await generateVoiceover(asset.script);
  const videoPath = await renderTrapVideo(audioPath, asset.script);

  await database.run(
    "UPDATE generated_assets SET audio_path = ?, video_path = ? WHERE id = ?",
    audioPath, videoPath, asset.id
  );

  return Response.json({ audioPath, videoPath });
}

import { renderMedia } from "@remotion/renderer";
import { bundle } from "@remotion/bundler";

export async function renderTrapVideo(audioPath, script) {
  const serveUrl = await bundle({
    entryPoint: "./src/video/TrapRoyaltiesReel.tsx",
  });

  const output = `./output/traproyalties_${Date.now()}.mp4`;

  await renderMedia({
    composition: "TrapRoyaltiesReel",
    serveUrl,
    codec: "h264",
    outputLocation: output,
    inputProps: { audioPath, script },
  });

  return output;
}
📁 /src/lib/elevenlabs.ts
ts
import fs from "fs";
import { ElevenLabsClient } from "@elevenlabs/elevenlabs-js";

const client = new ElevenLabsClient({
  apiKey: process.env.ELEVENLABS_API_KEY
});

export async function generateVoiceover(text) {
  const audio = await client.textToSpeech.convert("JBFqnCBsd6RMkjVDRZzb", {
    text,
    modelId: "eleven_multilingual_v2",
    outputFormat: "mp3_44100_128"
  });

  const path = `./output/voice_${Date.now()}.mp3`;
  fs.writeFileSync(path, Buffer.from(await audio.arrayBuffer()));

  return path;
}
